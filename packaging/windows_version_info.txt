# Windows resource block for Cura-Setup.exe.
#
# This does NOT stop SmartScreen. Nothing but a signature does. What it does is
# make the "More info" panel say "Cura Setup" and name a publisher, instead of
# showing a bare filename with every field blank -- which is what a scam looks
# like. If he gets as far as that panel, it should read like software.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(0, 1, 0, 0),
    prodvers=(0, 1, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '040904B0',
        [StringStruct('CompanyName', 'Liberation Labs / T.H. Coalition'),
         StringStruct('FileDescription', 'Cura Setup'),
         StringStruct('FileVersion', '0.1.0'),
         StringStruct('InternalName', 'Cura-Setup'),
         StringStruct('LegalCopyright',
                      'Hippocratic License 3.0 with AI Welfare module'),
         StringStruct('OriginalFilename', 'Cura-Setup.exe'),
         StringStruct('ProductName', 'Cura'),
         StringStruct('ProductVersion', '0.1.0')])
    ]),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
